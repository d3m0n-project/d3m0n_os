104static const char* str_findstr(const char* start, const char* end, const char* needle)
105{
106 size_t needlelen;

108 assert(start <= end, 0);
109
110 needlelen= strlen(needle);
111 assert(0 < needlelen, 0);
112 first = (unsigned char)needle[0];

115 {
116 const char* it= (const char*)memchr(start, first, (end - start) - (needlelen - 1));

272static sxmlerr_t parse_characters(sxml_t* state, sxml_args_t* args, const char* end)
273{
274 const char* start= buffer_fromoffset(args, state->bufferpos);
275 const char* limit, *colon, *ampr= str_findchr(start, end, '&');
276 assert(end <= buffer_getend (args), (sxmlerr_t)0);
277
278 if (ampr != start)
279 state_pushtoken(state, args, SXML_CHARACTER, start, ampr);
280
281 if (ampr == end)
282 return state_setpos(state, args, ampr);
283
284 /* limit entity to search to ENTITY_MAXLEN */
285 limit= MIN(ampr + ENTITY_MAXLEN, end);
286 colon= str_findchr(ampr, limit, ';');
287 if (colon == limit)
288 return (limit == end) ? SXML_ERROR_BUFFERDRY : SXML_ERROR_XMLINVALID;
289
290 start= colon + 1;
291 state_pushtoken(state, args, SXML_CHARACTER, ampr, start);
292 return state_setpos(state, args, start);
293}

307static sxmlerr_t parse_attributes(sxml_t* state, sxml_args_t* args)
308{
309 const char* start= buffer_fromoffset(args, state->bufferpos);
310 const char* end= buffer_getend(args);
311 const char* name= str_ltrim(start, end);
312
313 unsigned ntokens= state->ntokens;
314 assert(0 < ntokens, (sxmlerr_t)0);
315
316 while (name != end && ISALPHA(*name))
317 {
318 const char* eq, *space, *quot, *value;
319 sxmlerr_t err;
320
321 /* Attribute name */
322 eq= str_findchr(name, end, '=');
323 if (eq == end)
324 return SXML_ERROR_BUFFERDRY;
325
326 space= str_rtrim(name, eq);
327 state_pushtoken(state, args, SXML_CDATA, name, space);
328
329 /* Attribute value */
330 quot= str_ltrim(eq + 1, end);
331 if (quot == end)
332 return SXML_ERROR_BUFFERDRY;
333 else if (*quot != '\'' && *quot != '"')
334 return SXML_ERROR_XMLINVALID;
335
336 value= quot + 1;
337 quot= str_findchr(value, end, *quot);
338 if (quot == end)
339 return SXML_ERROR_BUFFERDRY;
340
341 state_setpos (state, args, value);
342 err= parse_attrvalue(state, args, quot);
343 if (err != SXML_SUCCESS)
344 return err;
345
346 /* --- */
347
348 name= str_ltrim(quot + 1, end);
349 }
350
351 {
352 sxmltok_t* token= args->tokens + (ntokens - 1);
353 token->size= (unsigned short) (state->ntokens - ntokens);
354 }
355
356 return state_setpos(state, args, name);

552static inline sxmlerr_t sxml_parse(sxml_t *state, const char *buffer, unsigned bufferlen, sxmltok_t tokens[], unsigned num_tokens)
553{
554 sxml_t temp= *state;
555 const char* end= buffer + bufferlen;
556
557 sxml_args_t args;
558 args.buffer= buffer;
559 args.bufferlen= bufferlen;
560 args.tokens= tokens;
561 args.num_tokens= num_tokens;
562
563 /* --- */
564
565 while (!ROOT_FOUND(&temp))
566 {
567 sxmlerr_t err;
568 const char* start= buffer_fromoffset(&args, temp.bufferpos);
569 const char* lt= str_ltrim(start, end);
570 state_setpos(&temp, &args, lt);
571 state_commit(state, &temp);
572
573 if (end - lt < TAG_MINSIZE)
574 return SXML_ERROR_BUFFERDRY;
575
576 /* --- */
577
578 if (*lt != '<')
579 return SXML_ERROR_XMLINVALID;
580
581 switch (lt[1])
582 {
583 case '?': err= parse_instruction(&temp, &args); break;
584 case '!': err= parse_doctype(&temp, &args); break;
585 default: err= parse_start(&temp, &args); break;
586 }
587
588 if (err != SXML_SUCCESS)
589 return err;
590
591 state_commit(state, &temp);
592 }
593
594 /* --- */
595
596 while (!ROOT_PARSED(&temp))
597 {
598 sxmlerr_t err;
599 const char* start= buffer_fromoffset(&args, temp.bufferpos);
600 const char* lt= str_findchr(start, end, '<');
601 while (buffer_fromoffset(&args, temp.bufferpos) != lt)
602 {
603 sxmlerr_t err= parse_characters(&temp, &args, lt);
604 if (err != SXML_SUCCESS)
605 return err;
606
607 state_commit(state, &temp);
608 }
609
610 /* --- */
611
612 if (end - lt < TAG_MINSIZE)
613 return SXML_ERROR_BUFFERDRY;
614
615 switch (lt[1])
616 {
617 case '?': err= parse_instruction(&temp, &args); break;
618 case '/': err= parse_end(&temp, &args); break;
619 case '!': err= (lt[2] == '-') ? parse_comment(&temp, &args) : parse_cdata (&temp, &args); break;
620 default: err= parse_start(&temp, &args); break;
621 }
622
623 if (err != SXML_SUCCESS)
624 return err;
625
626 state_commit(state, &temp);
627 }
628
629 return SXML_SUCCESS;
630}
